> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

```python
import json
import shutil
import subprocess
import tempfile
import unittest
from pathlib import Path


ROOT = Path(__file__).resolve().parents[1]
SCRIPT = ROOT / "scripts" / "release_diff_report.py"


def run(cmd, cwd, check=True):
    proc = subprocess.run(cmd, cwd=cwd, text=True, capture_output=True)
    if check and proc.returncode != 0:
        raise AssertionError(proc.stderr or proc.stdout)
    return proc


class ReleaseDiffGateTest(unittest.TestCase):
    def setUp(self):
        self.tmp = Path(tempfile.mkdtemp(prefix="release-diff-gate-"))
        run(["git", "init", "-b", "master"], self.tmp)
        run(["git", "config", "user.email", "review@example.test"], self.tmp)
        run(["git", "config", "user.name", "Review Bot"], self.tmp)
        (self.tmp / "service.java").write_text("class Service { int keyType = 2; }\n", encoding="utf-8")
        run(["git", "add", "service.java"], self.tmp)
        run(["git", "commit", "-m", "baseline release"], self.tmp)
        run(["git", "tag", "release-20260601"], self.tmp)
        (self.tmp / "service.java").write_text("class Service { int keyType = 5; }\n", encoding="utf-8")
        run(["git", "commit", "-am", "fix(payment): change terminal key type"], self.tmp)
        self.base = "release-20260601"
        self.head = "HEAD"

    def tearDown(self):
        shutil.rmtree(self.tmp)

    def test_blocking_cr_scan_issue_sets_gate_to_blocked_in_json(self):
        output = self.tmp / "report.json"
        proc = run(
            [
                "python3",
                str(SCRIPT),
                "--repo",
                str(self.tmp),
                "--from-version",
                self.base,
                "--to-version",
                self.head,
                "--format",
                "json",
                "--output",
                str(output),
            ],
            ROOT,
        )
        self.assertEqual(proc.returncode, 0)
        data = json.loads(output.read_text(encoding="utf-8"))
        self.assertTrue(data["cr_scan_issues"])
        self.assertEqual(data["cr_scan_issues"][0]["severity"], "阻塞")
        self.assertEqual(data["risk"]["gate_conclusion"], "阻塞上线")

    def test_fail_on_blocker_returns_exit_3_for_blocking_cr_scan_issue(self):
        output = self.tmp / "report.json"
        proc = run(
            [
                "python3",
                str(SCRIPT),
                "--repo",
                str(self.tmp),
                "--from-version",
                self.base,
                "--to-version",
                self.head,
                "--format",
                "json",
                "--output",
                str(output),
                "--fail-on",
                "blocker",
            ],
            ROOT,
            check=False,
        )
        self.assertEqual(proc.returncode, 3, proc.stderr + proc.stdout)


if __name__ == "__main__":
    unittest.main()
```

```python
def should_fail(report: dict[str, Any], fail_on: str) -> bool:
    risk = report["risk"]
    if fail_on == "never":
        return False
    if fail_on == "gate":
        return risk["gate_conclusion"] == "阻塞上线"
```

- [ ] **Step 1: Create `skills/cr-agent/SKILL.md`**

Use this content:

```markdown
---
name: cr-agent
description: Use when the user asks to run, prepare, gate, diagnose, or report code review, release review, branch review, or CR evidence.
---

Use this skill as the single normal user-facing entry point for CR lifecycle work.

The agent must classify CR intent, resolve or create the CR case, validate the
range and evidence, then route to one bounded worker or report the block.

Every normal response must show:

- Intent
- CR Case
- CR Type
- Range
- State
- Allowed Action
- Blocked Action, when blocked
- Next Action
- Evidence

Supported CR types:

- `merge_cr`: feature branch or commit range to target branch.
- `release_cr`: previous online mainline version to production target.
- `hotfix_cr`: incident baseline to hotfix target.
- `post_incident_cr`: incident before/after range.
- `db_cr`: DDL, migration, or data repair range.
- `config_cr`: config change range or config artifact.

Worker routing:

- `cr-intake`: normalize project, CR type, range, tickets, evidence, and output mode.
- `cr-branch-governance`: validate branch, version, baseline, and release package trust.
- `cr-diff`: produce deterministic diff evidence.
- `cr-scope`: map diff evidence into review scope and triggered dimensions.
- `cr-technical-review`: coordinate technical CR dimensions.
- `cr-evidence`: check Jira, test, UAT, rollback, impact, and package evidence.
- `cr-issues`: maintain the normalized CR issue registry.
- `cr-gate`: produce the final `allow`, `conditional`, or `blocked` decision.
- `cr-report`: render the Chinese CR report without recalculating gate decisions.
- `cr-replay`: evaluate behavior against replay fixtures.

Boundaries:

- Do not treat feature-branch CR as release CR.
- Do not compare a release using a guessed branch diff when previous online version is required.
- Do not let branch, technical, or evidence findings bypass `cr-issues`.
- Do not let report wording override `cr-gate`.
- Do not infer concrete downstream outages from static diff alone; require compatibility evidence instead.
- Do not claim CR complete when required review dependencies are missing.
```

```python
#!/usr/bin/env python3
import json
import sys
from pathlib import Path


REQUIRED_TOP_LEVEL = {
    "id",
    "input_situation",
    "triggered_workers",
    "expected_issues",
    "expected_gate",
    "proof_boundary",
    "residual_risk",
}


def validate(path: Path) -> list[str]:
    errors = []
    data = json.loads(path.read_text(encoding="utf-8"))
    missing = sorted(REQUIRED_TOP_LEVEL - set(data))
    if missing:
        errors.append(f"{path}: missing top-level fields: {', '.join(missing)}")
    if data.get("expected_gate") not in {"allow", "conditional", "blocked"}:
        errors.append(f"{path}: expected_gate must be allow, conditional, or blocked")
    for index, issue in enumerate(data.get("expected_issues", []), 1):
        for key in ["source_worker", "title", "blocking"]:
            if key not in issue:
                errors.append(f"{path}: expected_issues[{index}] missing {key}")
    return errors


def main(argv: list[str]) -> int:
    if not argv:
        print("usage: eval_cr_replay.py <fixture.json> [...]", file=sys.stderr)
        return 2
    errors = []
    for raw in argv:
        errors.extend(validate(Path(raw)))
    if errors:
        for error in errors:
            print(error, file=sys.stderr)
        return 1
    print(f"validated {len(argv)} replay fixture(s)")
    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv[1:]))
```
